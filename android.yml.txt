name: Build Android APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        variant: [standard, background]
    env:
      BG: ${{ matrix.variant == 'background' && '1' || '0' }}
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Separate app id for background variant
        if: matrix.variant == 'background'
        run: |
          sed -i 's/"com.jarvis.assistant"/"com.jarvis.assistant.bg"/; s/"appName": "Джарвис"/"appName": "Джарвис Фон"/' capacitor.config.json
          cat capacitor.config.json

      - name: Install dependencies
        run: npm install

      - name: Add Android platform
        run: npx cap add android

      - name: Permissions, icon, background service
        run: |
          pip install pillow || true
          python3 scripts/patch_android.py

      - name: Sync web files
        run: npx cap sync android

      - name: Build debug APK
        run: |
          cd android
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Name APK
        run: |
          if [ "${{ matrix.variant }}" = "background" ]; then N=Jarvis-Background.apk; else N=Jarvis.apk; fi
          cp android/app/build/outputs/apk/debug/app-debug.apk "$N"
          echo "APK_NAME=$N" >> "$GITHUB_ENV"

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: jarvis-apk-${{ matrix.variant }}
          path: ${{ env.APK_NAME }}

      - name: Publish release with APK
        continue-on-error: true
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh release create "apk-${{ matrix.variant }}-${{ github.run_number }}" "$APK_NAME" --title "Джарвис ${{ matrix.variant }} #${{ github.run_number }}" --notes "Скачай $APK_NAME на телефон и установи."
